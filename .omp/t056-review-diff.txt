+using System.Globalization;
+using Microsoft.AspNetCore.Http;
+
+namespace wa.api.Middleware;
+
+/// <summary>
+/// TA-4.1.6 CORS — only configured origins are allowed, no `*` wildcard (a wildcard would leak to any site). Config key
+/// <c>Wa:Cors:AllowedOrigins</c>: comma-separated list in prod, single value locally. Preflight answers carry a 1 h cache;
+/// unlisted origins get the plain pass-through (no CORS headers — the browser fails them), so non-browser traffic is unaffected.
+/// </summary>
+public sealed class CorsAllowListMiddleware
+{
+    private static readonly TimeSpan PreflightMaxAge = TimeSpan.FromHours(1); // TA-4.1.6: preflight cached 1 h
+
+    private readonly RequestDelegate _next;
+    private readonly string[] _allowedOrigins;
+
+    public CorsAllowListMiddleware(RequestDelegate next, IConfiguration configuration)
+    {
+        _next = next;
+        var configured = configuration["Wa:Cors:AllowedOrigins"];
+        // No config → CORS disabled (M0 local runs set the value via launchSettings; prod sets it per environment).
+        _allowedOrigins = string.IsNullOrWhiteSpace(configured)
+            ? []
+            : configured.Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries);
+    }
+
+    public async Task InvokeAsync(HttpContext context)
+    {
+        var origin = context.Request.Headers.Origin.ToString();
+        if (origin.Length > 0 && _allowedOrigins.Contains(origin, StringComparer.Ordinal))
+        {
+            context.Response.Headers.AccessControlAllowOrigin = origin; // echo the exact origin — never "*", TA-4.1.6
+            context.Response.Headers.AccessControlAllowCredentials = "true"; // httpOnly session cookie (TA-4.1.2) needs credentials
+            if (context.Request.Method == HttpMethods.Options) // preflight: 1 h cache so the browser stops re-asking per request
+            {
+                context.Response.Headers.Vary.Append("Origin");
+                context.Response.Headers.AccessControlAllowHeaders = "Authorization, Content-Type, Idempotency-Key";
+                context.Response.Headers.AccessControlAllowMethods = "GET, POST, PATCH, DELETE, OPTIONS";
+                context.Response.Headers.AccessControlMaxAge = PreflightMaxAge.TotalSeconds.ToString(CultureInfo.InvariantCulture); // StringValues has no implicit int conversion
+            }
+        }
+
+        await _next(context);
+    }
+}

+/// <summary>
+/// Second in the pipeline (after CorrelationMiddleware so its log context is active). Maps every failure to TA-4.1.3 Problem+JSON:
+/// <see cref="BusinessException"/> → its closed-list status + code (EC-056-2, never swallowed into 500), any other exception →

+    private readonly ILogger<GlobalErrorMiddleware> _logger;

+    {
+        _next = next;
+        _logger = logger;
+    }
+
+    public async Task InvokeAsync(HttpContext context)
+    {
+        try
+        {
+            await _next(context);
+        }
+        catch (BusinessException business) // EC-056-2: own code keeps its TA-4.1.3 status — the table in ProblemCodes is the single source of truth
+        {
+            if (!context.Response.HasStarted)
+                await ProblemWriter.WriteAsync(context, business.Status, business.Code, business.Message, business.Errors);
+            _logger.LogWarning("Business error {Code} (HTTP {Status}): {Message}", business.Code, business.Status, business.Message);
+            return;
+        }
+        catch (Exception unhandled) // non-business: 500 INTERNAL — stack stays in Serilog (TA-10.5), body gets the generic detail
+        {
+            _logger.LogError(unhandled, "Unhandled exception on {Path} — mapped to 500 INTERNAL", context.Request.Path);
+            if (!context.Response.HasStarted)
+                await ProblemWriter.WriteAsync(context, ProblemCodes.StatusOf(ProblemCodes.Internal), ProblemCodes.Internal, "An internal server error occurred.");

+        }
+

+        {
+            // Kestrel's HttpResponseStream doesn't support Position — the empty-body check only works under TestServer, so treat it as best-effort evidence.
+            var bodyIsEmpty = true;
+            try { bodyIsEmpty = context.Response.Body.Position == 0; } catch (NotSupportedException) { }
+            if (bodyIsEmpty)
+                await ProblemWriter.WriteAsync(context, ProblemCodes.StatusOf(ProblemCodes.NotFound), ProblemCodes.NotFound, "No route matches this request.");

+using System.Text.Json;
+using System.Text.Json.Nodes;
+using Microsoft.AspNetCore.Http;
+using wa.domain.Correlation;
+using wa.domain.Errors;
+
+namespace wa.api.Middleware;
+
+/// <summary>
+/// Serializes the exact TA-4.1.3 Problem+JSON field set — type, title, status, code, details, optional errors[], correlationId —
+/// in that order (field order per the RFC 7804 example). Status/title are pinned by <see cref="ProblemCodes"/> so callers cannot drift;
+/// correlationId comes from <see cref="CorrelationContext"/> (FR-056-1), never Activity.Current.
+/// </summary>
+public static class ProblemWriter
+{
+    /// <summary>TA-4.1.3 example format: base + lowercase machine code, e.g. <c>https://example.com/errors/transfer-size-exceeded</c>.</summary>
+    public const string TypeBase = "https://example.com/errors/";
+
+    /// <summary>Builds the Problem payload as a node tree with exact key order (JsonObject preserves insertion order).</summary>
+    public static JsonObject Build(int status, string code, string details, IReadOnlyList<ProblemFieldError>? errors = null)
+    {
+        var payload = new JsonObject()
+        {
+            ["type"] = TypeBase + code.ToLowerInvariant(), // e.g. https://example.com/errors/transfer-size-exceeded — exact TA-4.1.3 example value
+            ["title"] = ProblemCodes.TitleOf(code), // unknown codes fail fast here (ArgumentException at the call site)
+            ["status"] = status,
+            ["code"] = code,
+            ["details"] = details
+        };
+        if (errors is not null) // optional TA-4.1.3 errors[] — present only for VALIDATION-shaped responses; key order stays exact either way
+            payload["errors"] = new JsonArray(errors.Select(e => new JsonObject { ["field"] = e.Field, ["message"] = e.Message }).ToArray());
+        payload["correlationId"] = CorrelationContext.CorrelationId; // first 8 hex of the trace-id — always present (middleware generates when absent)
+
+        return payload;
+    }
+
+    /// <summary>
+    /// Writes the Problem response and commits it before writing a single body byte (Response.StartAsync). Kestrel flushes headers with the first bytes, so any header set after this method returns is lost — callers that must carry extra headers (e.g. Retry-After) set them on <c>context.Response.Headers</c> beforehand.
+    /// </summary>
+    public static async Task WriteCompleteAsync(HttpContext context, int status, string code, string details, IReadOnlyList<ProblemFieldError>? errors = null)
+    {
+        var payload = Build(status, code, details, errors);
+        context.Response.StatusCode = status;
+        context.Response.ContentType = "application/json";
+        await context.Response.StartAsync(); // commits headers — everything set on Response.Headers up to now is locked in
+        await JsonSerializer.SerializeAsync(context.Response.Body, payload);
+    }
+    public static async Task WriteAsync(HttpContext context, int status, string code, string details, IReadOnlyList<ProblemFieldError>? errors = null)
+    {
+        var payload = Build(status, code, details, errors);
+        context.Response.StatusCode = status;
+        context.Response.ContentType = "application/json";
+        await JsonSerializer.SerializeAsync(context.Response.Body, payload);
+    }
+}

+namespace wa.domain.Correlation;
+
+/// <summary>Validates incoming W3C <c>traceparent</c> headers (TA-10.1, FR-056-1). Malformed or wrong-version values are treated as absent (EC-056-1) — the pipeline never crashes on a bad header.</summary>
+public static class W3CTraceParser
+{
+    /// <summary>The only version this API understands at M0. Higher versions need an explicit upgrade decision (TA-10.1).</summary>
+    public const string SupportedVersion = "00";
+
+    private const int TraceIdLength = 32; // 16 bytes hex
+    private const int ParentIdLength = 16; // 8 bytes hex
+
+    /// <summary>Returns the validated trace-id (32 lowercase hex chars), or <c>null</c> when the header is absent/malformed/wrong-version.</summary>
+    public static string? ParseTraceId(string? traceparent)
+        => Parse(traceparent)?.TraceId;
+
+    /// <summary>Parses a W3C <c>traceparent</c> value. Returns the full 4-field form (version zero-padded to "00", lowercase), or <c>null</c>.</summary>
+    public static TraceParent? Parse(string? traceparent)
+    {
+        if (string.IsNullOrWhiteSpace(traceparent))
+            return null;
+
+        var parts = traceparent.Trim().Split('-'); // strict W3C form: exactly version-traceid-parentid-flags
+        if (parts.Length != 4 || !IsAllHex(parts[0], 2) || parts[0] != SupportedVersion
+            || !IsAllHex(parts[1], TraceIdLength) || !IsAllHex(parts[2], ParentIdLength)
+            || !IsAllHex(parts[3], 2)) // trace-flags must be a 2-hex byte (TA-4.1.3-style strictness at M0)
+            return null;
+
+        var traceId = parts[1].ToLowerInvariant();
+        if (traceId.All(c => c == '0')) // all-zero trace-id is invalid per W3C — treat as absent
+            return null;
+
+        return new TraceParent(parts[0], traceId, parts[2].ToLowerInvariant(), parts[3]);
+    }
+
+    /// <summary>Generates a fresh trace-id (16 random bytes) for requests without an inbound header.</summary>
+    public static string NewTraceId() => Guid.NewGuid().ToString("n");
+
+    private static bool IsAllHex(string s, int length) =>
+        s.Length == length && s.All(c => c is >= '0' and <= '9' or >= 'a' and <= 'f');
+}
+
+/// <summary>A validated W3C traceparent (version 00).</summary>
+public sealed record TraceParent(string Version, string TraceId, string ParentId, string Flags)
+{
+    /// <summary>The full header value — kept separately for App Insights (TA-10.1).</summary>
+    public string ToHeader() => $"{Version}-{TraceId}-{ParentId}-{Flags}";
+
+    /// <summary>CorrelationId = first 8 hex of trace-id (TA-10.1) — exactly 8 chars, not the full trace-id.</summary>
+    public string CorrelationId => TraceId[..8];
+}
